! =====================================================================
!  ROUTEUR Cisco 1941 - R-LaPlateforme
!  Rôle : passerelle Internet (NAT/PAT), routage inter-VLAN
!         (router-on-a-stick) et serveur DHCP pour les 4 VLAN.
! =====================================================================
!  Plan d'adressage (source de vérité = tableau d'adressage du sujet) :
!   VLAN 1  VoIP           192.168.0.0/24    passerelle .1   DHCP .10-.50
!   VLAN 10 PC fixes       192.168.10.0/24   passerelle .1   DHCP .10-.50
!   VLAN 20 Wi-Fi          192.168.20.0/24   passerelle .1   DHCP .10-.50
!   VLAN 30 Administration 192.168.30.0/24   passerelle .1   DHCP .10-.50
! =====================================================================

enable
configure terminal
!
hostname R-LaPlateforme
!
no ip domain-lookup
ip domain-name laplateforme.io
!
! ---------------------------------------------------------------------
!  Exclusions DHCP : on ne distribue QUE la plage .10 - .50 (cf. sujet)
!  -> on exclut .1-.9 (passerelle + réservations) et .51-.254
! ---------------------------------------------------------------------
ip dhcp excluded-address 192.168.0.1 192.168.0.9
ip dhcp excluded-address 192.168.0.51 192.168.0.254
ip dhcp excluded-address 192.168.10.1 192.168.10.9
ip dhcp excluded-address 192.168.10.51 192.168.10.254
ip dhcp excluded-address 192.168.20.1 192.168.20.9
ip dhcp excluded-address 192.168.20.51 192.168.20.254
ip dhcp excluded-address 192.168.30.1 192.168.30.9
ip dhcp excluded-address 192.168.30.51 192.168.30.254
!
! ---------------------------------------------------------------------
!  Pools DHCP (un par VLAN)
! ---------------------------------------------------------------------
ip dhcp pool VLAN1-VOIP
 network 192.168.0.0 255.255.255.0
 default-router 192.168.0.1
 dns-server 8.8.8.8
!
ip dhcp pool VLAN10-PCFIXES
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
!
ip dhcp pool VLAN20-WIFI
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8
!
ip dhcp pool VLAN30-ADMIN
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 8.8.8.8
!
! ---------------------------------------------------------------------
!  Interface LAN (G0/0) en trunk 802.1Q -> sous-interfaces par VLAN
!  = "router-on-a-stick" : assure le routage entre tous les VLAN.
! ---------------------------------------------------------------------
interface GigabitEthernet0/0
 no ip address
 duplex auto
 speed auto
 no shutdown
!
interface GigabitEthernet0/0.1
 description VLAN 1 - VoIP (VLAN natif)
 encapsulation dot1Q 1 native
 ip address 192.168.0.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/0.10
 description VLAN 10 - PC fixes
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/0.20
 description VLAN 20 - Wi-Fi
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/0.30
 description VLAN 30 - Administration
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip nat inside
!
! ---------------------------------------------------------------------
!  Interface WAN (G0/1) vers Internet - adresse fournie par le FAI (DHCP)
!  Une route par défaut est installée automatiquement via DHCP.
! ---------------------------------------------------------------------
interface GigabitEthernet0/1
 description Uplink Internet (WAN)
 ip address 8.8.8.1 255.255.255.0
 ip nat outside
 duplex auto
 speed auto
 no shutdown
!
! Dans le lab, le "monde Internet" est represente par un serveur 8.8.8.8/24
! directement relie a G0/1 (fait aussi office de DNS). Pour un vrai FAI en DHCP :
! interface GigabitEthernet0/1
!  ip address dhcp
!  ip nat outside
!
! ---------------------------------------------------------------------
!  NAT / PAT : tous les VLAN sortent sur Internet via l'IP du G0/1
! ---------------------------------------------------------------------
access-list 1 permit 192.168.0.0 0.0.0.255
access-list 1 permit 192.168.10.0 0.0.0.255
access-list 1 permit 192.168.20.0 0.0.0.255
access-list 1 permit 192.168.30.0 0.0.0.255
!
ip nat inside source list 1 interface GigabitEthernet0/1 overload
!
! ---------------------------------------------------------------------
!  Sécurité d'accès minimale
! ---------------------------------------------------------------------
enable secret Cisco_2025!
line con 0
 logging synchronous
!
end
write memory
